(*
Ce sujet a pour but de d'émontrer des propriétés de fonctions définies sur des arbres binaires. 
*)

(***************************)
(* Type des arbres binaires  *)
(***************************)

Inductive btree : Type :=
| Leaf : btree
| Node : btree -> nat -> btree -> btree.

(*********************)
(* Fonctions de base *)
(*********************)

Fixpoint size (t: btree) : nat :=
 match t with
 | Leaf => 0
 | Node l _ r => 1 + size l + size r
 end.

Fixpoint height (t: btree) : nat :=
 match t with
 | Leaf => 0
 | Node l _ r => 1 + max (height l) (height r)
 end.

(* parcours infixe dfe l'arbre *)
Fixpoint inorder (t: btree) : list nat :=
 match t with
 | Leaf => []
 | Node l x r => inorder l ++ [x] ++ inorder r
 end.

(* parcours préfixe de l'arbre *)
Fixpoint preorder (t: btree) : list nat :=
 match t with
 | Leaf => []
 | Node l x r => [x] ++ preorder l ++ preorder r
 end.

(* parcours postfixe de l'arbre *)
Fixpoint postorder (t: btree) : list nat :=
 match t with
 | Leaf => []
 | Node l x r => postorder l ++ postorder r ++ [x]
 end.

(* application d'une fonction sur chaque noeud de l'arbre *)
Fixpoint map_btree (f: nat -> nat) (t: btree) : btree :=
 match t with
 | Leaf => Leaf
 | Node l x r => Node (map_btree f l) (f x) (map_btree f r)
 end.

Fixpoint mirror (t: btree) : btree :=
 match t with
 | Leaf => Leaf
 | Node l x r => Node (mirror r) x (mirror l)
 end.

(* nombre d'occurrences d'une information dans l'arbre *)
Fixpoint count (n: nat) (t: btree) : nat :=
 match t with
 | Leaf => 0
 | Node l x r => (if Nat.eqb x n then 1 else 0) + count n l + count n r
 end.

(* Définition d'équilibre *)
Fixpoint is_balanced (t: btree) : bool :=
 match t with
 | Leaf => true
 | Node l _ r =>
     let hl := height l in
     let hr := height r in
     (hl <=? S hr) && (hr <=? S hl)
     && is_balanced l && is_balanced r
 end.
